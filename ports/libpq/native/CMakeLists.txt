# Translate the upstream autoheader inputs into configure_file templates.
# Undefined facilities stay undefined, matching PostgreSQL's optional API use.
foreach(header IN ITEMS pg_config.h pg_config_ext.h)
  file(READ "${src}/include/${header}.in" template)
  string(REGEX REPLACE "#undef ([A-Za-z_][A-Za-z_0-9]*)" "#cmakedefine \\1 @\\1@" template "${template}")
  file(WRITE "${gen}/${header}.in" "${template}")
  configure_file("${gen}/${header}.in" "${gen}/${header}" @ONLY)
endforeach()
configure_file("${src}/include/port/win32.h" "${gen}/pg_config_os.h" COPYONLY)

function(pg_client_library target directory)
  set(sources ${ARGN})
  list(TRANSFORM sources PREPEND "${src}/${directory}/")
  list(TRANSFORM sources APPEND .c)
  add_library(${target} ${sources})
  target_link_libraries(${target} PRIVATE pgcommon pgport pg_frontend)
  target_include_directories(${target} PRIVATE "${src}/interfaces/ecpg/include")
  if(BUILD_SHARED_LIBS)
    file(STRINGS "${src}/${directory}/exports.txt" exports REGEX "^[A-Za-z_]+[A-Za-z_0-9]*[ \t]+[0-9]+")
    set(def "EXPORTS\n")
    foreach(symbol IN LISTS exports)
      string(REGEX REPLACE "^([A-Za-z_0-9]+)[ \t]+([0-9]+).*" "\\1 @\\2" symbol "${symbol}")
      string(APPEND def "${symbol}\n")
    endforeach()
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${target}.def" "${def}")
    target_sources(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${target}.def")
  endif()
  install(TARGETS ${target} RUNTIME DESTINATION bin ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
endfunction()

function(pg_install_catalogs directory domain)
  file(GLOB translations CONFIGURE_DEPENDS "${src}/${directory}/po/*.po")
  set(catalogs)
  foreach(po IN LISTS translations)
    get_filename_component(locale "${po}" NAME_WE)
    set(destination "${CMAKE_CURRENT_BINARY_DIR}/locale/${locale}/LC_MESSAGES")
    set(mo "${destination}/${domain}.mo")
    add_custom_command(OUTPUT "${mo}"
      COMMAND "${CMAKE_COMMAND}" -E make_directory "${destination}"
      COMMAND "${PG_MSGFMT}" -o "${mo}" "${po}"
      DEPENDS "${po}" VERBATIM)
    list(APPEND catalogs "${mo}")
    install(FILES "${mo}" DESTINATION "share/libpq/locale/${locale}/LC_MESSAGES")
  endforeach()
  add_custom_target(catalogs_${domain} ALL DEPENDS ${catalogs})
endfunction()
